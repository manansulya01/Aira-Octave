from pathlib import Path

p = Path("kernel/task/task.c")
s = p.read_text()

start = s.index("static uint64_t *build_initial_stack(")
end = s.index("\nvoid task_init(void)", start)

new = r'''static uint64_t *build_initial_stack(
    uint8_t *stack,
    void (*entry)(void)
)
{
    /*
     * irq_common expects this exact layout at RSP:
     *
     *   r15
     *   r14
     *   r13
     *   r12
     *   r11
     *   r10
     *   r9
     *   r8
     *   rbp
     *   rdi
     *   rsi
     *   rdx
     *   rcx
     *   rbx
     *   rax
     *   vector
     *   error
     *   RIP
     *   CS
     *   RFLAGS
     *
     * Because the task runs in the same ring (ring 0) as the
     * kernel, the CPU does NOT push RSP/SS for the interrupt.
     *
     * irq_common eventually executes:
     *
     *   pop rax ... pop r15
     *   add rsp, 16
     *   iretq
     *
     * Therefore the fake frame must exactly match that layout.
     */

    uint64_t *top =
        (uint64_t *)(stack + TASK_STACK_SIZE);

    uint64_t *sp = top;

    /*
     * Build backwards so the final memory layout is:
     *
     *   r15 ... rax
     *   vector
     *   error
     *   RIP
     *   CS
     *   RFLAGS
     */

    /* CPU IRET frame */
    *--sp = 0x202;             /* RFLAGS */
    *--sp = 0x08;              /* CS */
    *--sp = (uint64_t)entry;   /* RIP */

    /* Software-added interrupt metadata */
    *--sp = 0;                 /* error */
    *--sp = 0;                 /* vector */

    /* Saved general-purpose registers */
    *--sp = 0;                 /* rax */
    *--sp = 0;                 /* rbx */
    *--sp = 0;                 /* rcx */
    *--sp = 0;                 /* rdx */
    *--sp = 0;                 /* rsi */
    *--sp = 0;                 /* rdi */
    *--sp = 0;                 /* rbp */
    *--sp = 0;                 /* r8 */
    *--sp = 0;                 /* r9 */
    *--sp = 0;                 /* r10 */
    *--sp = 0;                 /* r11 */
    *--sp = 0;                 /* r12 */
    *--sp = 0;                 /* r13 */
    *--sp = 0;                 /* r14 */
    *--sp = 0;                 /* r15 */

    return sp;
}
'''

p.write_text(s[:start] + new + s[end:])

print("task.c build_initial_stack() fixed successfully")

run_step() {
    local NAME="$1"
    shift
    local OUT="$LOGDIR/${NAME}.log"

    echo
    echo "------------------------------------------------------------------"
    echo "STEP: $NAME"
    echo "------------------------------------------------------------------"

    set +e
    "$@" >"$OUT" 2>&1

    cat "$OUT"

    if [ "$RC" -eq 0 ]; then
        echo "[PASS] $NAME"
    else
        echo "[FAIL] $NAME (exit $RC)"
    fi

    return 0
}
